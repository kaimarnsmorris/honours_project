% --- S0: overview / roadmap (the gist in one slide) ---
\begin{frame}{Overview}
  \textbf{Neural Bayes estimators} learn the map from a dataset straight to its posterior estimate --- giving fast, \emph{amortised} Bayesian inference.
  \vspace{0.15em}
  \begin{itemize}
    \item \textbf{The catch:} training one is \emph{itself} a Monte Carlo problem --- it needs \alert{many simulated samples}, and each gradient step is noisy.
    \item \textbf{This thesis:} treat that as \textbf{variance reduction} --- cut the noise and train faster, on \alert{fewer samples}.
  \end{itemize}
  \vspace{0.25em}
  \begin{columns}[t]
    \begin{column}{0.47\textwidth}
      \begin{block}{Idea 1 $\cdot$ Rao--Blackwellisation}
        Replace noisy targets by their exact conditional mean. \textbf{Provably} lower variance --- needs a \emph{conjugate} block.
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{block}{Idea 2 $\cdot$ Importance sampling}
        Draw from a proposal and reweight. \textbf{General} --- but with \emph{no guarantee}.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% --- S1a: Bayesian inference refresher ---
\begin{frame}{Bayesian inference}
  Everything we do sits within a \textbf{Bayesian inference} framework.
  \begin{itemize}
    \item Fix a model: parameters $\bt$ with prior $p(\bt)$, and data $\by$ with likelihood $p(\by\given\bt)$.
    \item Seeing $\by$, Bayes' rule updates the prior into the \textbf{posterior distribution}:
      \[ p(\bt\given\by)\;\propto\; p(\by\given\bt)\,p(\bt). \]
    \item The posterior is our full, updated belief about $\bt$.
  \end{itemize}
  \vspace{0.2em}
  \begin{center}
  \begin{tikzpicture}[font=\scriptsize]
    \draw[->] (-3,0) -- (3.2,0) node[right]{$\bt$};
    \draw[gray, thick] plot[smooth] coordinates {(-3,0.05)(-2,0.16)(-1,0.29)(0,0.35)(1,0.29)(2,0.16)(3,0.05)};
    \node[gray] at (-2.15,0.35){prior};
    \draw[accent, very thick] plot[smooth] coordinates {(-0.6,0.03)(0,0.18)(0.4,0.5)(0.8,0.92)(1.2,0.5)(1.6,0.18)(2.2,0.03)};
    \node[accent] at (1.9,0.85){posterior};
    \draw[accent, dashed] (0.8,0) -- (0.8,0.9);
    \fill[accent] (0.8,0) circle (1.3pt);
    \node[accent, anchor=north, font=\scriptsize] at (0.8,-0.04) {$\E[\bt\given\by]$};
  \end{tikzpicture}
  \end{center}
\end{frame}

\begin{frame}{Point estimation}
  \begin{itemize}
    \item Often we do not want the whole posterior --- just a single \textbf{point estimate} $\hat{\bt}$ of $\bt$.
    \item The usual choice is the \textbf{posterior mean} $\hat{\bt}=\E[\bt\given\by]$ --- in principle a \textbf{deterministic mapping} from the data $\by$ to $\hat{\bt}$.
    \item But it is \alert{rarely available in closed form} --- so we need a \textbf{procedure} that turns $\by$ into $\hat{\bt}$:
  \end{itemize}

% --- S1c: the Bayes risk (decision theory) --- what makes an estimate optimal ---
\begin{frame}{The Bayes risk}
  Which point estimate $\hat{\bt}(\by)$ should we return? Fix a \textbf{loss} $\ell(\bt,\hat{\bt})$ and minimise the \textbf{Bayes risk}
  \[ \E_{\bt,\by}\big[\ell\big(\bt,\hat{\bt}(\by)\big)\big]. \]
  The choice of loss sets \emph{which} posterior summary is optimal:
  \begin{center}
  \renewcommand{\arraystretch}{1.1}
  \begin{tabular}{lc}
    \toprule
    loss $\ell(\bt,\hat{\bt})$ & optimal estimate (its minimiser) \\
    \midrule
    squared \ $\lVert\bt-\hat{\bt}\rVert^2$ & posterior \textbf{mean} $\;\E[\bt\given\by]$ \\
    absolute \ $\lvert\bt-\hat{\bt}\rvert$ & posterior \textbf{median} \\
    quantile \ $(\alpha-\mathbf{1}_{\{\bt<\hat{\bt}\}})(\bt-\hat{\bt})$ & posterior \textbf{quantile} \\
    \bottomrule
  \end{tabular}
  \end{center}
  \vspace{-0.1em}
  {\footnotesize Usually this criterion is used to \emph{choose} which estimator to use.}
\end{frame}

% --- The idea: training is Monte Carlo (back to the Bayes risk) ---
\begin{frame}{Training a neural Bayes estimator}
  \textbf{Goal:} choose the network weights $\bg$ that minimise
  \[ L(\bg)=\E_{\bt,\by}\big[\ell\big(\bt,\net(\by)\big)\big]. \]
  No closed form --- so each step estimates it by \textbf{Monte Carlo} on a minibatch and takes a \textbf{stochastic-gradient} step:
  \begin{align*}
    \widehat{L}(\bg) &= \frac{1}{B}\sum_{b=1}^{B}\ell\big(\bt^{[b]},\net(\by^{[b]})\big),
      \qquad (\bt^{[b]},\by^{[b]})\sim\text{model}, \\[0.3em]
    \bg^{\mathrm{new}} &= \bg - \eta\,\nabla_{\bg}\widehat{L}(\bg).
  \end{align*}

  \vspace{0.6em}
  {\footnotesize \alert{Insight:} the minibatch gradient $\nabla_{\bg}\widehat{L}$ is a \textbf{random estimate} --- it has variance, so we can \alert{apply variance-reduction techniques} to speed up and stabilise training, and save simulation cost.}
\end{frame}

\section{Idea 1: Rao--Blackwellisation}
% ===================================================================

% --- S7: the trick (Rao-Blackwellise the minibatch loss estimator) ---
\begin{frame}{The trick}
  The training target $\bt$ is a \textbf{noisy draw}. Split $\bt=(\bt_1,\bt_2)$; in the minibatch loss estimator, replace each term by its \textbf{conditional mean} given $(\bt_2^{[b]},\by^{[b]})$ --- averaging out the noise in $\bt_1$:
  \begin{align*}
    \widehat{L}_{\MC}(\bg) &=\frac{1}{B}\sum_{b=1}^{B}\ell\big(\bt^{[b]},\net(\by^{[b]})\big), \\[0.2em]
    \widehat{L}_{\RB}(\bg) &=\frac{1}{B}\sum_{b=1}^{B}\E\big[\,\ell\big(\bt^{[b]},\net(\by^{[b]})\big)\,\big|\,\bt_2^{[b]},\by^{[b]}\,\big].
  \end{align*}
  \begin{itemize}
    \item Same mean --- but the \textbf{Rao--Blackwell theorem} makes $\widehat{L}_{\RB}$ have \alert{never larger variance}, for any model and any architecture.
    \item Exactly computable when the block $\bt_1$ is \textbf{conditionally conjugate}.
  \end{itemize}
\end{frame}

% --- S8: squared-error loss makes the conditional mean explicit ---
\begin{frame}{Squared error: a clean target}
  For \textbf{squared-error} loss the conditional mean is simple --- expand each term of $\widehat{L}_{\RB}$:
  \[
    \widehat{L}_{\RB}(\bg)=\frac{1}{B}\sum_{b=1}^{B}\Big[\;
      \big\lVert \net(\by^{[b]})-{\color{accent}\E[\bt\given\bt_2^{[b]},\by^{[b]}]}\big\rVert^2
      \;+\; \underbrace{\Tr\!\big(\Var(\bt\given\bt_2^{[b]},\by^{[b]})\big)}_{\text{constant in }\bg}\;\Big].
  \]
  \begin{itemize}
    \item Just squared error to a \textbf{clean target} ${\color{accent}\E[\bt\given\bt_2^{[b]},\by^{[b]}]}$ --- the exact conditional mean, in place of the noisy draw $\bt^{[b]}$.
    \item The variance term is \textbf{constant in $\bg$}: it drops straight out of the gradient.
  \end{itemize}
  \vspace{0.2em}
  {\footnotesize \textbf{Bayesian linear regression:} $\bt=(\bb,\sigma^2)$ with Gaussian prior $\bb\sim\Normal(0,\sigma_0^2 I)$; condition on $\sigma^2$ and $\bb$ is Gaussian, so the clean target $\E[\bb\given\sigma^2,\by]$ is \textbf{closed-form}.}
\end{frame}

\section{Idea 2: Importance sampling}
% ===================================================================

% --- S10: importance sampling — the idea (empirical IS estimator) ---
\begin{frame}{Importance sampling: A more general lever}
  Another commonly used variance-reduction technique is \textbf{importance sampling} --- and, unlike Rao--Blackwell, it needs \textbf{no conjugacy}.
  \[
    \widehat{L}_{\mathrm{IS}}(\bg)=\frac{1}{B}\sum_{b=1}^{B} {\color{accent}w(\bt^{[b]})}\,\ell\big(\bt^{[b]},\net(\by^{[b]})\big),
    \qquad \bt^{[b]}\sim q,\quad {\color{accent}w(\bt)=\tfrac{p(\bt)}{q(\bt)}}.
  \]
  \begin{itemize}
    \item Draw the parameters from a \textbf{proposal} $q(\bt)$ instead of the prior $p(\bt)$, and \textbf{reweight} each draw.
    \item Unbiased for \alert{any} proposal $q$ --- the weight $w=p/q$ undoes the change of measure.
    \item In general, a \textbf{careful choice of $q$} can \alert{lower the variance}.
    \item But \alert{no guarantee} --- a poor proposal \emph{inflates} the variance instead.
  \end{itemize}
\end{frame}

\section{Conclusion}
% ===================================================================

% --- S12: conclusion (results + significance + punchline) ---
\begin{frame}{Conclusion}
  \textbf{Goal:} improve training for neural Bayes estimators by more efficient sampling of the Bayes risk.
  \begin{itemize}
    \item \textbf{Rao--Blackwellisation} (needs a conjugate block): a \emph{provably} lower-variance training gradient, for any model or architecture --- and can be applied wherever Gibbs sampling is used.
    \item \textbf{Importance sampling} (needs nothing special): more general, but unguaranteed and problem-dependent --- here it did not help.
  \end{itemize}
  \vspace{0.5em}
  \alert{\textbf{Future work:}} many more variance-reduction techniques remain to be tried --- particularly \textbf{control variates}.
\end{frame}
