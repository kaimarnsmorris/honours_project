% --- S1: Bayesian setup (box v1) ---
% --- S1a: Bayesian inference refresher ---
\begin{frame}{Bayesian inference}
  \begin{itemize}
    \item Fix a model: parameters $\bt$ with prior $p(\bt)$, and data $\by$ with likelihood $p(\by\given\bt)$.
    \item Seeing $\by$, Bayes' rule updates the prior into the \textbf{posterior distribution}:
      \[ p(\bt\given\by)\;\propto\; p(\by\given\bt)\,p(\bt). \]
    \item The posterior is our full, updated belief about $\bt$.
  \end{itemize}
  \vspace{0.2em}
  \begin{center}
  \begin{tikzpicture}[font=\scriptsize]
    \draw[->] (-3,0) -- (3.2,0) node[right]{$\theta$};
    \draw[gray, thick] plot[smooth] coordinates {(-3,0.05)(-2,0.16)(-1,0.29)(0,0.35)(1,0.29)(2,0.16)(3,0.05)};
    \node[gray] at (-2.15,0.35){prior};
    \draw[accent, very thick] plot[smooth] coordinates {(-0.6,0.03)(0,0.18)(0.4,0.5)(0.8,0.92)(1.2,0.5)(1.6,0.18)(2.2,0.03)};

    \draw[accent, dashed] (0.8,0) -- (0.8,0.9);
    \fill[accent] (0.8,0) circle (1.3pt);
    \node[accent, anchor=north, font=\scriptsize] at (0.8,-0.04) {$\E[\bt\given\by]$};
  \end{tikzpicture}
  \end{center}
\end{frame}

    \item But it is \alert{rarely available in closed form} --- so we need a \textbf{procedure} that turns $\by$ into $\hat{\bt}$:
  \end{itemize}

% --- S5: loss functions pick the summary (last Background slide) ---
\begin{frame}{The loss picks which summary you estimate}
  The network minimises the \textbf{Bayes risk} $\;L(\bg)=\E_{\bt,\by}\big[\ell\big(\bt,\net(\by)\big)\big]$ --- and the loss $\ell$ decides \emph{which} posterior summary it targets:
  \begin{center}
  \renewcommand{\arraystretch}{1.15}
  \begin{tabular}{lc}
    \toprule
    loss $\ell(\bt,\hat{\bt})$ & best estimate (its minimiser) \\
    \midrule
    squared \ $\lVert\bt-\hat{\bt}\rVert^2$ & posterior \textbf{mean} $\;\E[\bt\given\by]$ \\
    absolute \ $\lvert\bt-\hat{\bt}\rvert$ & posterior \textbf{median} \\
    quantile \ $(\alpha-\mathbf{1}_{\{\bt<\hat{\bt}\}})(\bt-\hat{\bt})$ & posterior \textbf{quantile} \\
    \bottomrule
  \end{tabular}
  \end{center}
  \vspace{0.1em}
  In training we \textbf{approximate} this expectation by \textbf{Monte Carlo}:
  \[
    L(\bg)\;\approx\;\tfrac{1}{N}\sum_{i=1}^{N}\ell\big(\bt_i,\net(\by_i)\big),\quad (\bt_i,\by_i)\sim\text{model}.
  \]
\end{frame}

% --- The idea: training is Monte Carlo (last Background slide) ---
\begin{frame}{The idea}
  In training we estimate the risk by \textbf{Monte Carlo}:
  \[
    L(\bg)\;\approx\;\tfrac{1}{N}\sum_{i=1}^{N}\ell\big(\bt_i,\net(\by_i)\big),\qquad (\bt_i,\by_i)\sim\text{model}.
  \]
  \begin{itemize}
    \item So the \textbf{gradient} $g(\bg)=\nabla_{\bg} L(\bg)$ that drives training is a \alert{random quantity with variance}.
    \item That variance is about \emph{how we sample the loss}, not the inference problem --- \alert{so we can reduce it}.
  \end{itemize}
  \vspace{0.3em}
  {\footnotesize \emph{What if}, for part of the parameters, we could compute a conditional expectation like $\E[\bt_1\given\bt_2,\by]$ in closed form? \alert{That is the first lever.}}
\end{frame}

\section{Idea 1: Rao--Blackwellisation}
% ===================================================================

% --- S7: Rao-Blackwell idea ---
\begin{frame}{The trick}
  A classical trick for killing Monte Carlo noise.
  \begin{itemize}
    \item Estimating an average by sampling? If part of it can be computed \textbf{exactly}, do that --- replace the sampled piece by its exact \textbf{conditional mean}.
    \item Conditioning-and-averaging can only \emph{remove} variance, never add it.
  \end{itemize}
  \vspace{0.6em}
  \begin{center}\small
  noisy draws \; $\longrightarrow$ \; replace one block by its exact mean \; $\longrightarrow$ \; \alert{tighter estimate}
  \end{center}
  \vspace{0.5em}
  {\footnotesize Named after the Rao--Blackwell theorem.}
\end{frame}

% --- S8: RB on the loss + the guarantee (total-variance on slide + bar) ---
\begin{frame}{Rao--Blackwellising the loss --- and why it must help}
  Split $\bt=(\tau,\bb)$ with $\bb$ conditionally conjugate; integrate $\bb$ out by replacing its sampled target with $\E[\bb\given\tau,\by]$. For the gradient $g$,
  \[
    \Var(g)\;=\;\underbrace{\Var\big(\E[g\given\tau,\by]\big)}_{\Var(g_{\RB})}\;+\;\underbrace{\E\big[\Var(g\given\tau,\by)\big]}_{\succeq\,0}.
  \]
  \begin{columns}[c]
    \begin{column}{0.62\textwidth}
      \begin{itemize}
        \item Law of total variance $\Rightarrow\ \Var(g_{\RB})\preceq\Var(g)$.
        \item \alert{Never increases gradient variance} --- any model, any architecture.
        \item Linear net: the reduction reaches the fitted weights too.
      \end{itemize}
    \end{column}
    \begin{column}{0.36\textwidth}
      \centering
      \begin{tikzpicture}[font=\scriptsize]
        \fill[blue!30] (0,0) rectangle (0.75,2.2);
        \node[font=\tiny] at (0.375,-0.28){standard};
        \fill[orange!55] (1.6,0) rectangle (2.35,1.15);
        \node[font=\tiny] at (1.975,-0.28){RB};
        \draw[<->] (2.6,1.15) -- (2.6,2.2);
        \node[font=\tiny,anchor=west] at (2.62,1.68){removed};
        \node[rotate=90,font=\tiny] at (-0.35,1.1){gradient variance};
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

\section{Idea 2: importance sampling}
% ===================================================================

% --- S10: importance sampling idea ---
\begin{frame}{A more general lever}
  \textbf{No conjugacy required} --- unlike Rao--Blackwell.
  \begin{itemize}
    \item Simulate from a \textbf{proposal} that oversamples the informative regions, then \textbf{reweight} back to the original risk (unbiased for any proposal).
    \item But \alert{no guarantee}: bad weights can \emph{increase} variance and shrink the effective sample size.
  \end{itemize}
  \vspace{0.3em}
  \begin{center}
    \begin{tikzpicture}[font=\scriptsize,>=Stealth]

\section{Conclusion}
% ===================================================================

% --- S12: conclusion (results + significance + punchline) ---
\begin{frame}{Conclusion}
  Training a neural Bayes estimator is itself Monte Carlo --- so its training variance can be reduced.
  \begin{itemize}
    \item \textbf{Rao--Blackwellisation} (needs a conjugate block): a \emph{provable} reduction --- closes $\sim$half the gap to the Bayes floor and matches the standard estimator at $\tfrac14$ the batch size.
    \item \textbf{Importance sampling} (needs nothing special): more general, but unguaranteed and problem-dependent --- here it did not help.
  \end{itemize}
  \vspace{0.5em}
  \begin{center}\alert{\textbf{Conjugacy --- the classical trick behind Gibbs sampling --- is just as useful for \emph{training} the estimators meant to replace it.}}\end{center}
\end{frame}
