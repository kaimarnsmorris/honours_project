\section{Background: neural Bayes estimators}
% ===================================================================

% --- S1: Bayesian setup (box v1) ---
\begin{frame}{Bayesian inference: the setup}
  \begin{itemize}
    \item Fix a model: parameters $\bt$ with prior $p(\bt)$, data $\by$ with likelihood $p(\by\given\bt)$.
    \item Seeing $\by$, Bayes' rule gives the \textbf{posterior} $p(\bt\given\by)\propto p(\by\given\bt)\,p(\bt)$; we want a \textbf{summary}, usually the mean $\hat{\bt}=\E[\bt\given\by]$.
    \item Rarely available in closed form --- so we need a \textbf{procedure} that turns $\by$ into $\hat{\bt}$:
  \end{itemize}
  \vspace{0.6em}
  \begin{center}\estbox{dashed, draw=gray}{{\Large\color{gray}\bfseries ?}}\end{center}
  \vspace{0.6em}
  \begin{center}\alert{The next two slides fill this box in two different ways.}\end{center}
\end{frame}

% --- S2: MCMC (box v2) ---
\begin{frame}{Filling the box (1): Markov chain Monte Carlo}
  Sample the posterior, then average:
  \[
    \bt^{(1)},\dots,\bt^{(S)}\sim p(\bt\given\by),\qquad
    \E[\bt\given\by]\approx \tfrac{1}{S}\textstyle\sum_{s}\bt^{(s)}.
  \]
  \begin{center}\estbox{fill=black!8}{MCMC sampler}\end{center}
  \vspace{0.3em}
  \begin{itemize}
    \item Accurate and general --- but must be \alert{re-run from scratch for every new $\by$}, and can be slow.
    \item Do inference for \emph{many} datasets and you pay that cost \emph{every time}.
  \end{itemize}
\end{frame}

% --- S3: NBE + amortisation (box v3 + cost plot) ---
\begin{frame}{Filling the box (2): a neural network}
  Make the box a \textbf{neural network} $\net(\by)$, trained \emph{once} --- a \emph{neural Bayes estimator}.
  \begin{itemize}
    \item Training data is free: \textbf{simulate} pairs $(\bt,\by)$ from the model, and adjust $\bg$ so $\net(\by)\approx\bt$.
    \item Then every new dataset is a single \textbf{forward pass} --- the training cost is \alert{amortised}.
  \end{itemize}
  \vspace{0.2em}
  \begin{columns}[c]
    \begin{column}{0.55\textwidth}
      \centering\estbox{fill=black!8}{network $\net$}
    \end{column}
    \begin{column}{0.43\textwidth}
      \centering
      \begin{tikzpicture}[font=\scriptsize,>=Stealth]

  \end{columns}
  \vspace{0.2em}
  {\footnotesize \alert{Pay once} up front; after that, inference is nearly free.}
\end{frame}

\section{Our contribution: variance reduction}
% ===================================================================

% --- S5: Bayes risk + squared-error => posterior mean ---
\begin{frame}{What is the network trained to do?}
  It minimises the \textbf{Bayes risk} --- expected loss over the model:
  \[
    L(\bg)=\E_{\bt,\by}\big[\,\ell\big(\bt,\net(\by)\big)\,\big].
  \]
  \begin{itemize}
    \item Under squared error, the minimiser is exactly the \textbf{posterior mean}:
      \[
        \arg\min_{f}\ \E\big[\lVert\bt-f(\by)\rVert^2\big]=\E[\bt\given\by].
      \]
      \alert{That is why a well-trained network approximates $\E[\bt\given\by]$} --- it explains the last slide.
    \item Change the loss, change the summary: absolute error $\to$ median, quantile loss $\to$ quantiles.
  \end{itemize}
  \vspace{0.1em}
  {\footnotesize Pointwise in $\by$, the best constant $c$ minimising $\E[\lVert\bt-c\rVert^2\given\by]$ is $c=\E[\bt\given\by]$.}
\end{frame}

% --- S6: the pivot — training is Monte Carlo ---
\begin{frame}{The observation this thesis starts from}
  We never have the true risk --- we estimate it by \textbf{Monte Carlo} over simulated draws:
  \[
    L(\bg)\;\approx\;\tfrac{1}{N}\sum_{i=1}^{N}\big\lVert\bt_i-\net(\by_i)\big\rVert^2,
    \qquad (\bt_i,\by_i)\ \text{simulated}.
  \]
  \begin{itemize}
    \item So the loss --- and the \textbf{gradient} that drives training --- is a \alert{random quantity with variance}.
    \item That variance comes from \emph{how we sample the loss}, not from the inference problem.
    \item \alert{So we can reduce it}, with the classical variance-reduction tools of Monte Carlo.
  \end{itemize}
\end{frame}

% --- S7: Rao-Blackwell idea ---
\begin{frame}{Idea 1: Rao--Blackwellisation}
  A classical trick for killing Monte Carlo noise.
  \begin{itemize}
    \item Estimating an average by sampling? If part of it can be computed \textbf{exactly}, do that --- replace the sampled piece by its exact \textbf{conditional mean}.
    \item Conditioning-and-averaging can only \emph{remove} variance, never add it.
  \end{itemize}
  \vspace{0.6em}
  \begin{center}\small
  noisy draws \; $\longrightarrow$ \; replace one block by its exact mean \; $\longrightarrow$ \; \alert{tighter estimate}
  \end{center}
  \vspace{0.5em}
  {\footnotesize Named after the Rao--Blackwell theorem.}
\end{frame}

% --- S8: RB on the loss + the guarantee (total-variance on slide + bar) ---
\begin{frame}{Rao--Blackwellising the loss --- and why it must help}
  Split $\bt=(\tau,\bb)$ with $\bb$ conditionally conjugate; integrate $\bb$ out by replacing its sampled target with $\E[\bb\given\tau,\by]$. For the gradient $g$,
  \[
    \Var(g)\;=\;\underbrace{\Var\big(\E[g\given\tau,\by]\big)}_{\Var(g_{\RB})}\;+\;\underbrace{\E\big[\Var(g\given\tau,\by)\big]}_{\succeq\,0}.
  \]
  \begin{columns}[c]
    \begin{column}{0.62\textwidth}
      \begin{itemize}
        \item Law of total variance $\Rightarrow\ \Var(g_{\RB})\preceq\Var(g)$.
        \item \alert{Never increases gradient variance} --- any model, any architecture.
        \item Linear net: the reduction reaches the fitted weights too.
      \end{itemize}
    \end{column}
    \begin{column}{0.36\textwidth}
      \centering
      \begin{tikzpicture}[font=\scriptsize]
        \fill[blue!30] (0,0) rectangle (0.75,2.2);
        \node[font=\tiny] at (0.375,-0.28){standard};
        \fill[orange!55] (1.6,0) rectangle (2.35,1.15);
        \node[font=\tiny] at (1.975,-0.28){RB};
        \draw[<->] (2.6,1.15) -- (2.6,2.2);

        \node[rotate=90,font=\tiny] at (-0.35,1.1){gradient variance};
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

% --- S10: importance sampling idea ---
\begin{frame}{Idea 2: importance sampling}
  A more general lever --- \textbf{no conjugacy required}.
  \begin{itemize}
    \item Simulate from a \textbf{proposal} that oversamples the informative regions, then \textbf{reweight} back to the original risk (unbiased for any proposal).
    \item But \alert{no guarantee}: bad weights can \emph{increase} variance and shrink the effective sample size.
  \end{itemize}
  \vspace{0.3em}
  \begin{center}

\section{Conclusion}
% ===================================================================

% --- S12: conclusion (results + significance + punchline) ---
\begin{frame}{Conclusion}
  Training a neural Bayes estimator is itself Monte Carlo --- so its training variance can be reduced.
  \begin{itemize}
    \item \textbf{Rao--Blackwellisation} (needs a conjugate block): a \emph{provable} reduction --- closes $\sim$half the gap to the Bayes floor and matches the standard estimator at $\tfrac14$ the batch size.
    \item \textbf{Importance sampling} (needs nothing special): more general, but unguaranteed and problem-dependent --- here it did not help.
  \end{itemize}
  \vspace{0.5em}
  \begin{center}\alert{\textbf{Conjugacy --- the classical trick behind Gibbs sampling --- is just as useful for \emph{training} the estimators meant to replace it.}}\end{center}
\end{frame}
